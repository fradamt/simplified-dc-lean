> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the one fixed FG module `SimplexFG.fj` as pure Lean definitions, and prove its static FG lemmas in `Kit.FG`. The module is DC's FG without SG and grades, with the FJ root rule in place of NJ ("DC-fj"). Each pinned lemma names one `docs/FG-COMPOSITION.md` id, and each pin records its coverage clause by clause.

**Architecture:** There are two new Lake libraries in `lean/available-chain`:
- `AvailableChainSimplexFG` (LM, definitions only): FG rows and blocks, the chain state and transition with no NJ heights, the E1/E2 evidence, the FG signing client with its record guards, the DC-fj store (one cumulative admitted tree `TF`, the spec's `Σ.T`, with the refusal `on_block` and the active view; the live caches; `update_finality` in DC form), the FJ controls, the filtered tree and the FJ frame.
- `AvailableChainKitFG` (L5, proofs): the ported DC certificate lemmas and entry geometry, `fg.accountable.v1`, `fg.uniqueJust.v1`, `fg.finalLock.v1`, the client record safety, the store invariant, the store controls (the frame source of ALFS) and the frame lemmas, provenance, the causal-cap core, `fj.crossingNoFinal.v1`, `fj.inBandLock.v1`, `fj.frameFeasible.v1`, and the `fj.twoChanges.core.v1` result.

Code is PORTED (copied and adapted) from verified-consensus. It is not imported across Lake projects. The module does not import the framework, the engine or any AP. It is FG-only: the AP (for example SA) enters later through the framework interface (FRAMEWORK-DESIGN §7.3).

**Tech Stack:** Lean 4 v4.30.0-rc2 and Mathlib v4.30.0-rc2. Both `available-chain` and `decoupled-consensus/verified-consensus` have `lean-toolchain` = `leanprover/lean4:v4.30.0-rc2`. The build uses Lake and the module system.

## Review Focus

In the uncontested or gap frame of the same blocks, `lo = H−1`. r148 §2 and the spec say that the code must change both viability and below-band eligibility (r148 N12). Eligibility and the walk are AP machinery and are not in this module; the AP obligation is `ap.headReachesLo.v1` in the view form and `ap.headReachesLo.v5` (clauses 1–3) under ALFS (with the anchor fallback: a late contest can empty the frozen domain). abbrev Height := Nat

```lean
inductive NamedHeightPair where
  | empty
  | vote (height : Height) (entry : BlockId) (timeout : Bool)
  deriving DecidableEq, Repr

namespace NamedHeightPair
/-- Targets and timeouts use the same height and entry match. -/
def matchesEntry (height : Height) (entry : BlockId) : NamedHeightPair → Bool
  | .empty => false
  | .vote h target _ => decide (h = height ∧ target = entry)
end NamedHeightPair

/-- The finality pair `(h_f, T_f)` (port: DC/DecoupledConsensusModel/Objects/Blocks.lean:60–66). -/
structure FinalityPair where
  height : Height
  target : BlockId
  deriving DecidableEq, Repr

/-- One FG row: signer, height pair, finality pair. No SG field. -/
structure Row (V : Type) where
  signer : V
  height_pair : NamedHeightPair
  finality_pair : Option FinalityPair
  deriving DecidableEq, Repr

/-- A block as the FG reads it. The AP maps its blocks to this type later. -/
inductive FGBlock (V : Type) where
  | genesis
  | node (parent : FGBlock V) (slot : Slot) (root : BlockId) (rows : List (Row V))
  deriving DecidableEq, Repr

namespace FGBlock
variable {V : Type}
def slot : FGBlock V → Slot
  | .genesis => 0
  | .node _ s _ _ => s
def root : FGBlock V → BlockId
  | .genesis => genesisRoot
  | .node _ _ r _ => r
def parent : FGBlock V → FGBlock V
  | .genesis => .genesis
  | .node p _ _ _ => p
def rows : FGBlock V → List (Row V)
  | .genesis => []
  | .node _ _ _ rs => rs

variable [DecidableEq V]
def preceq (A : FGBlock V) : FGBlock V → Bool
  | .genesis => decide (A = .genesis)
  | B@(.node p _ _ _) => decide (A = B) || preceq A p
def Preceq (A B : FGBlock V) : Prop := preceq A B = true
instance (A B : FGBlock V) : Decidable (Preceq A B) :=
  inferInstanceAs (Decidable (preceq A B = true))
def compatible (A B : FGBlock V) : Bool := preceq A B || preceq B A
end FGBlock

/-- The rows carried by the chain that ends at `B` (port: DC/DecoupledConsensusModel/Protocol/Evidence.lean `chain_attestations`). -/
def chainRows {V : Type} [DecidableEq V] : FGBlock V → Finset (Row V)
  | .genesis => ∅
  | .node p _ _ rs => rs.toFinset ∪ chainRows p

/-- Fixed positive weights (port: DC/DecoupledConsensusModel/Objects/Weights.lean:49–78). FG-SA uses weight 1. -/
structure Electorate (V : Type) [Fintype V] where
  weight : V → Nat
  weight_pos : ∀ i, 0 < weight i

namespace Electorate
variable {V : Type} [Fintype V] [DecidableEq V]
def totalWeight (E : Electorate V) : Nat := ∑ i, E.weight i
def weightOf (E : Electorate V) (S : Finset V) : Nat := ∑ i ∈ S, E.weight i
/-- `q = ⌈2W/3⌉`; with unit weights this is FG-SA's `q_F = (2n+2) div 3`. -/
def q (E : Electorate V) : Nat := (2 * E.totalWeight + 2) / 3
def IsQuorum (E : Electorate V) (S : Finset V) : Prop := E.q ≤ E.weightOf S
end Electorate
```

```lean
def lo (c : Controls V) : Height := if c.contest then c.H else c.H - 1
```

```lean
/-- The manifest that `scripts/check-fg-pins.sh` checks, from the elaborated
values (Astra P01, Q01–Q03), one item per line: `src <key> <root> <path>
<sha256>` for each source of the pins and `defPins` (without duplicates),
`pin <id>`, `ap <id>` and `external <id>` for each active reference, and
`notFormalized <id>` for each historical entry. The last line is the trailer
`END src=<n> pin=<n> ap=<n> external=<n> notFormalized=<n>`. -/
def emitManifest (ps : List Pin) (ds : List DefPin) (nf : List (String × String)) :
    IO Unit := do
  let srcs := (ps.flatMap (·.sources) ++ ds.map (·.src)).eraseDups
  let refs := ps.flatMap fun p => p.clauses.filterMap fun c =>
    match c.cov with
    | .ap i _ => some ("ap", i)
    | .external i _ => some ("external", i)
    | _ => none
  for s in srcs do IO.println s!"src {s.key} {s.root} {s.path} {s.sha256}"
  for p in ps do IO.println s!"pin {p.id}"
  for (t, i) in refs do IO.println s!"{t} {i}"
  for (i, _) in nf do IO.println s!"notFormalized {i}"
  let n (t : String) := (refs.filter (·.1 == t)).length
  IO.println (s!"END src={srcs.length} pin={ps.length} ap={n "ap"} " ++
    s!"external={n "external"} notFormalized={nf.length}")

/-- Every Lean name of the table (the pins, `defPins` and `helpers`), without
duplicates, for the axiom check of Step 3. -/
def pinNames (ps : List Pin) (ds : List DefPin) (hs : List Lean.Name) : List Lean.Name :=
  (ps.flatMap fun p => p.clauses.flatMap fun c =>
    match c.cov with
    | .lean ns => ns
    | .leanHyp ns _ => ns
    | _ => []) ++ ds.map (·.lean) ++ hs |>.eraseDups
```

```bash
# Checks the ELABORATED pin table of AvailableChainKitFG/Pins.lean against
# docs/FG-COMPOSITION.md. It reads only the manifest that
# scripts/FGPinIds.lean prints (never the Lean text):
#   src <key> <root> <path> <sha256>, pin <id>, ap <id>, external <id>,
#   notFormalized <id>, and the last line
#   END src=<n> pin=<n> ap=<n> external=<n> notFormalized=<n>.

#    last line and occurs once, each count equals the number of lines with
#    its tag, and there is at least one src and one pin (else exit 2);
# 1) each src resolves to its REGISTERED version (tsc/: the archive copy by
#    hash; spec: the archived file; vc and dc: the registered Git commit), and
#    that version has the pinned hash;

#    tables (a continuation line counts only when its key column is blank);
# 3) each id is, literally, the id of an entry heading;
# 4) each id has exactly one row in the "## Index" table, and for pin, ap and
#    external ids the status there does not start with SUPERSEDED.
#    notFormalized ids are historical and are exempt from the status test.
# OUT=<file> reads a saved manifest instead of running FGPinIds (tests).

    *) return 1 ;;
  esac
}
[ -s "$REG" ] || { echo "MISSING OR EMPTY: $REG"; exit 2; }
tmp=$(mktemp -d); trap 'rm -rf "$tmp"' EXIT
if [ -n "${OUT:-}" ]; then
  [ -s "$OUT" ] || { echo "MISSING OR EMPTY: $OUT"; exit 2; }

  grep -Eq '^END src=[0-9]+ pin=[0-9]+ ap=[0-9]+ external=[0-9]+ notFormalized=[0-9]+$' ||
  { echo "NO END TRAILER (truncated manifest)"; exit 2; }

  want=$(printf '%s\n' "$last" | tr ' ' '\n' | awk -F= -v t="$tag" '$1 == t { print $2 }')
  have=$(grep -c "^$tag " "$tmp/body" || true)
  [ "$want" -eq "$have" ] || { echo "COUNT MISMATCH: $tag trailer $want lines $have"; exit 2; }
done

npin=$(grep -c '^pin ' "$tmp/body" || true)

  case $st in SUPERSEDED*) echo "SUPERSEDED: $tag $id ($st)"; status=1 ;; esac
done < "$tmp/ids"
nref=$(grep -Ec '^(ap|external) ' "$tmp/ids" || true)
nhist=$(grep -c '^notFormalized ' "$tmp/ids" || true)
echo "checked $nsrc sources, $npin pin ids, $nref active references and $nhist historical ids"
exit $status
```
