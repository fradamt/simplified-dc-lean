## Review Focus

1. **A variant mix-up.** A theorem that is proved for `nj` is used as if it holds for `fj`. FG-COMPOSITION §4 says that HNI, trunk and one bullet do NOT apply to FJ. Test: Task 16's pin table lists a variant for each pinned lemma. A `check` example fails if a `fj`-pinned lemma mentions `njFrame` or an `nj`-pinned lemma mentions `fjFrame`.
2. **"Compatible with F" against "F ⪯ J".** `fg.frame.v1` takes the greatest justified pair "compatible with F_d". DC's `update_finality` and `fj.frame.v1` require `F ⪯ J`. The two agree only when F's own justification carrier is in the view (the retained F basis). Test: Task 7 states the correspondence with an explicit `FBasisIn V F` hypothesis, and has a `decide` counterexample without it.
3. **Natural-number subtraction in `H−1`.** `Height = ℕ` truncates `H − 1` at `H = 0` or `H = 1`. The frames use `H ≥ 1`. Test: Task 6 has `decide` examples at `H = 1`, with a view that holds only genesis.
4. **The lex order on `(h_j, J.root)`.** FJ takes the lex-greatest `(σ[W].h_j, root(σ[W].J))`. Ties at one height are broken by root. Test: Task 6 has an example with two justified entries at one height, and checks that the lex-greater root wins.
5. **An empty view or a view without a height-`H` witness.** `D` can be empty, and the walk then returns the root. Test: Task 6 checks `fjFrame` and `njFrame` on `{genesis}`: `H = 1`, `Z = genesis`, `D = {genesis}`.

```lean
open AvailableChain

abbrev Height := Nat

/-- The height pair of a row, entry-bound (port: DC/DecoupledConsensusModel/Objects/NamedBlocks.lean:31–48). -/
inductive NamedHeightPair where
  | empty
  | vote (height : Height) (entry : BlockId) (timeout : Bool)
  deriving DecidableEq, Repr

namespace NamedHeightPair
/-- Both targets and timeouts use the same height and entry match. -/
def matchesEntry (height : Height) (entry : BlockId) : NamedHeightPair → Bool
  | .empty => false
  | .vote h target _ => decide (h = height ∧ target = entry)
end NamedHeightPair

/-- The finality pair `(h_f, T_f)` (port: DC/…/Objects/Blocks.lean:60–66). -/
structure FinalityPair where
  height : Height
  target : BlockId
  deriving DecidableEq, Repr

/-- One FG row: signer, height pair, finality pair. No SG field. -/
structure Row (V : Type) where
  signer : V
  height_pair : NamedHeightPair
  finality_pair : Option FinalityPair
  deriving DecidableEq, Repr

/-- A block as the FG reads it: parent, slot, root and the carried rows.
The AP maps its own blocks to this type (Kit.Compose, later). -/
inductive FGBlock (V : Type) where
  | genesis
  | node (parent : FGBlock V) (slot : Slot) (root : BlockId) (rows : List (Row V))
  deriving DecidableEq, Repr

namespace FGBlock
variable {V : Type}
def slot : FGBlock V → Slot
  | .genesis => 0
  | .node _ s _ _ => s
def root : FGBlock V → BlockId
  | .genesis => genesisRoot
  | .node _ _ r _ => r
def parent : FGBlock V → FGBlock V
  | .genesis => .genesis
  | .node p _ _ _ => p
def rows : FGBlock V → List (Row V)
  | .genesis => []
  | .node _ _ _ rs => rs

variable [DecidableEq V]
def preceq (A : FGBlock V) : FGBlock V → Bool
  | .genesis => decide (A = .genesis)
  | B@(.node p _ _ _) => decide (A = B) || preceq A p
def Preceq (A B : FGBlock V) : Prop := preceq A B = true
instance (A B : FGBlock V) : Decidable (Preceq A B) :=
  inferInstanceAs (Decidable (preceq A B = true))
def compatible (A B : FGBlock V) : Bool := preceq A B || preceq B A
end FGBlock

/-- The rows carried by the chain that ends at `B` (port: DC/…/Protocol/Evidence.lean `chain_attestations`). -/
def chainRows {V : Type} [DecidableEq V] : FGBlock V → Finset (Row V)
  | .genesis => ∅
  | .node p _ _ rs => rs.toFinset ∪ chainRows p

/-- Fixed positive weights (port: DC/…/Objects/Weights.lean:49–78). FG-SA uses weight 1. -/
structure Electorate (V : Type) [Fintype V] where
  weight : V → Nat
  weight_pos : ∀ i, 0 < weight i

namespace Electorate
variable {V : Type} [Fintype V] [DecidableEq V]
def totalWeight (E : Electorate V) : Nat := ∑ i, E.weight i
def weightOf (E : Electorate V) (S : Finset V) : Nat := ∑ i ∈ S, E.weight i
/-- `q = ⌈2W/3⌉`. With unit weights this is FG-SA's `q_F = (2n+2) div 3`. -/
def q (E : Electorate V) : Nat := (2 * E.totalWeight + 2) / 3
def IsQuorum (E : Electorate V) (S : Finset V) : Prop := E.q ≤ E.weightOf S
end Electorate
```

  H : Height
  j : Height
  J : FGBlock V
  Z : FGBlock V
  lo : Height

  Vw.image (fun W => ((derive E cfg W).h_j, (derive E cfg W).J))

/-- The lex key of a pair. -/
def key (p : Height × FGBlock V) : Lex (ℕ × ℕ) := toLex (p.1, p.2.root.value)

/-- Lex-greatest `(h_j, J.root)` among the pairs that satisfy `ok`; `(0, genesis)`
if none. `Finset.fold` needs a commutative operation, so take the greatest KEY
with `Finset.sup` (`Lex (ℕ × ℕ)` is a linear order with `⊥`), then the pair with
that key. Ties in the key (same height, same root) are the same block under root
injectivity; without it, `pickPair` returns one of them (port `pickUnique?` and
`find?` from DC/DecoupledConsensusModel/Objects/Blocks.lean:290–333). -/
def lexMax (ok : FGBlock V → Bool) (Vw : View V) : Height × FGBlock V :=
  let cands := (pairs E cfg Vw).filter (fun p => ok p.2)
  let k := cands.sup key
  ((cands.filter (fun p => key p = k)).image id).min' ?_ -- see note

compatible with F; `Z = J` iff `H = j+1`; the band starts at `H−1`. -/
def njFrame (F : FGBlock V) (Vw : View V) : Frame V :=
  let H := frontier E cfg Vw
  let (j, J) := lexMax E cfg (fun J => FGBlock.compatible F J) Vw
  let Z := if H = j + 1 then J else F
  let D := Vw.filter (fun B => FGBlock.preceq F B && FGBlock.preceq Z B &&
    decide (∃ W ∈ Vw, FGBlock.Preceq B W ∧ H - 1 ≤ hOf E cfg W))
  ⟨F, H, j, J, Z, H - 1, D, false⟩

/-- [pin fj.frame.v1] The FJ frame (r148 §2): `F ⪯ J`; a contest at `H = j+1`
when a frontier block does not extend `J`; then root `F` and band `{H}`. -/
def fjFrame (F : FGBlock V) (Vw : View V) : Frame V :=
  let H := frontier E cfg Vw
  let (j, J) := lexMax E cfg (fun J => FGBlock.preceq F J) Vw
  let contest := decide (H = j + 1) &&
    decide (∃ W ∈ Vw, hOf E cfg W = H ∧ ¬ FGBlock.Preceq J W)
  let Z := if H = j + 1 ∧ contest = false then J else F
  let lo := if contest then H else H - 1
  let viable := Vw.filter (fun B => FGBlock.preceq F B &&
    decide (∃ W ∈ Vw, FGBlock.Preceq B W ∧ lo ≤ hOf E cfg W))
  ⟨F, H, j, J, Z, lo, viable.filter (fun B => FGBlock.preceq Z B), contest⟩

`h_max`, `h_j` and `J` from the view (FRAMEWORK-DESIGN §7.2). -/
def r1Store (F : FGBlock V) (Vw : View V) : DCStore V :=
  let (j, J) := lexMax E cfg (fun J => FGBlock.preceq F J) Vw
